# Playbook v1 (Claude, 2026-09-27): the scripted teacher's gate-proven tactics (TEACHER.md, sf2/teacher.py),
# rewritten as note-field rules. The teacher's source rule is in each reason. What the note cannot express:
# the apex kick timing (no jump height in the note), the 40 px throw range (the note's "close" is < 80 px),
# and whether he is closing in (dx_trend). Ground rules carry me_state=stand so the air rule wins in the air.

# anti-air (teacher rule 11): he jumps at her; fierce up close, roundhouse further out
opp_state=jump dist=close me_state=stand -> hp            # anti-air up close: fierce (rule 11)
opp_state=jumpattack dist=close me_state=stand -> hp      # anti-air up close: fierce (rule 11)
opp_state=jump dist=mid me_state=stand -> hk              # anti-air at mid range: roundhouse (rule 11)
opp_state=jumpattack dist=mid me_state=stand -> hk        # anti-air at mid range: roundhouse (rule 11)

# guard (teacher rule 1): crouch-guard his attacks and a fireball close to her
opp_state=attack dist=close me_state=stand -> block       # he attacks within reach: crouch-guard (rule 1)
opp_state=attack dist=mid me_state=stand -> block         # he attacks at mid range: crouch-guard (rule 1)
fireball=close me_state=stand -> block                    # a hadouken is about to hit: crouch-guard (rule 1)

# throw (teacher rule 8): up close against a standing or crouching Ryu; at 40-80 px this becomes a fierce
opp_state=stand dist=close me_state=stand -> throw        # walk-in throw (rule 8)
opp_state=crouch dist=close me_state=stand -> throw       # walk-in throw (rule 8)

# jump in (teacher rule 5): from mid range when he is not attacking; the air rule does the kicking
opp_state=stand dist=mid me_state=stand -> jump_forward   # jump in over his limbs (rule 5)
opp_state=crouch dist=mid me_state=stand -> jump_forward  # jump in over his limbs (rule 5)

# in the air (teacher rule 7): kick; the timing (near the apex) is laya's to find
me_state=jump -> hk                                       # jump-in kick (rule 7, timing not expressible)

# far away: close the distance
dist=far me_state=stand -> forward                        # walk in from far (teacher default)
